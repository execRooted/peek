# ls + custom logic based on the value of _HVIEW_ECHO.
__hview_ls() {
    # shellcheck disable=SC2164
    \command ls "$@" {%- if echo %} && __hview_pwd {%- endif %}
}

# Wrapper for ls to automatically add listed directories to hview.
__hview_ls_wrapper() {
    \command ls "$@" || return $?
    if [ "$#" -ge 1 ]; then
        i=$#
        while [ "$i" -gt 0 ]; do
            # shellcheck disable=SC2016
            eval "arg=\${$i}"
            if [ -d "$arg" ]; then
                \command hview add -- "$arg"
                break
            fi
            i=$((i - 1))
        done
    fi
}
alias ls=__hview_ls_wrapper

# Hook to add new entries to the database.
__hview_hook() {
    \command hview add -- "$(__hview_pwd || \command true)"
}

# Initialize hook.
if [ "${PS1:=}" = "${PS1#*\$(__hview_hook)}" ]; then
    PS1="${PS1}\$(__hview_hook)"
fi

# Report common issues.
__hview_doctor() {
{%- if hook != InitHook::Prompt %}
    return 0
{%- else %}
    [ "${_HVIEW_DOCTOR:-1}" -eq 0 ] && return 0
    case "${PS1:-}" in
    *__hview_hook*) return 0 ;;
    *) ;;
    esac

    _HVIEW_DOCTOR=0
    \command printf '%s\n' \
        'hview: detected a possible configuration issue.' \
        'Please ensure that hview is initialized right at the end of your shell configuration file.' \
        '' \
        'If the issue persists, consider filing an issue at:' \
        'https://github.com/execRooted/hview/issues' \
        '' \
        'Disable this message by setting _HVIEW_DOCTOR=0.' \
        '' >&2
{%- endif %}
}

# Jump to a directory using only keywords.
__hview_h() {
    __hview_doctor

    if [ "$#" -eq 0 ]; then
        \command ls ~
    elif [ "$#" -eq 1 ] && [ "$1" = '-' ]; then
        if [ -n "${OLDPWD}" ]; then
            \command ls "${OLDPWD}"
        else
            # shellcheck disable=SC2016
            \command printf 'hview: $OLDPWD is not set'
            return 1
        fi
    elif [ "$#" -eq 1 ] && (\command cd -- "$1") >/dev/null 2>&1; then
        \command ls "$1"
    else
        __hview_result="$(\command hview query --exclude "$(__hview_pwd || \command true)" -- "$@")" &&
            \command ls "${__hview_result}"
    fi
}

# Jump to a directory using interactive search.
__hview_hi() {
    __hview_doctor
    __hview_result="$(\command hview query --interactive -- "$@")" && \command ls "${__hview_result}"
}
