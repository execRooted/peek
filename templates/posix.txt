if \command -v cygpath >/dev/null

  __peek_pwd() {

  }

  __peek_pwd() {
    {{ pwd }}
  }

# ls + custom logic based on the value of _PEEK_ECHO.
__peek_ls() {
    # shellcheck disable=SC2164
    \command ls "$@" {%- if echo %} && __peek_pwd {%- endif %}
}

{% match hook %}

{%- when InitHook::Prompt -%}
# Hook to add new entries to the database.
__peek_hook() {
    \command peek add -- "$(__peek_pwd || \command true)"
}

# Initialize hook.
if [ "${PS1:=}" = "${PS1#*\$(__peek_hook)}" ]; then
    PS1="${PS1}\$(__peek_hook)"
fi

# Report common issues.
__peek_doctor() {
{%- if hook != InitHook::Prompt %}
    return 0
{%- else %}
    [ "${_PEEK_DOCTOR:-1}" -eq 0 ] && return 0
    case "${PS1:-}" in
    *__peek_hook*) return 0 ;;
    *) ;;
    esac

    _PEEK_DOCTOR=0
    \command printf '%s\n' \
        'peek: detected a possible configuration issue.' \
        'Please ensure that peek is initialized right at the end of your shell configuration file.' \
        '' \
        'If the issue persists, consider filing an issue at:' \
        'https://github.com/execRooted/peek/issues' \
        '' \
        'Disable this message by setting _PEEK_DOCTOR=0.' \
        '' >&2
{%- endif %}
}

# Jump to a directory using only keywords.
__peek_p() {
    __peek_doctor

    if [ "$#" -eq 0 ]; then
        ls ~
    elif [ "$#" -eq 1 ] && [ "$1" = '-' ]; then
        if [ -n "${OLDPWD}" ]; then
            ls "${OLDPWD}"
        else
            # shellcheck disable=SC2016
            \command printf 'peek: $OLDPWD is not set'
            return 1
        fi
    elif [ "$#" -eq 1 ] && (\command cd -- "$1") >/dev/null 2>&1; then
        ls "$1"
    else
        __peek_result="$(\command peek query --exclude "$(__peek_pwd || \command true)" -- "$@")" &&
            ls "${__peek_result}"
    fi
}

# Jump to a directory using interactive search.
__peek_pi() {
    __peek_doctor
    __peek_result="$(\command peek query --interactive -- "$@")" && ls "${__peek_result}"
}
