# Jump to a directory using only keywords.
fn __hview_h {|@rest|
    if (builtin:eq [] $rest) {
        ls .
    } elif (builtin:eq [-] $rest) {
        ls $oldpwd
    } elif (and ('builtin:==' (builtin:count $rest) 1) (path:is-dir &follow-symlink=$true $rest[0])) {
        ls $rest[0]
        hview add -- $rest[0]
    } else {
        var path
        try {
            set path = (hview query --list --exclude $pwd -- $@rest)
        } catch {
        } else {
            if (!= $path "") {
                set path_count = (builtin:count (str:split "\n" $path))
                if (== $path_count 1) {
                    ls $path
                    hview add -- $path
                } else {
                    set path = (hview query --interactive --exclude $pwd -- $@rest)
                    ls $path
                }
            }
        }
    }
}
edit:add-var __hview_h~ $__hview_h~

# Jump to a directory using interactive search.
fn __hview_hi {|@rest|
    var path
    try {
        set path = (hview query --interactive -- $@rest)
    } catch {
    } else {
        ls $path
        hview add -- $path
    }
}
edit:add-var __hview_hi~ $__hview_hi~

# Load completions.
{#-
  hview-based completions are currently not possible, because Elvish only prints
  a completion if the current token is a prefix of it.
#}
fn __hview_h_complete {|@rest|
    if (!= (builtin:count $rest) 2) {
        builtin:return
    }
    edit:complete-filename $rest[1] |
        builtin:each {|completion|
            var dir = $completion[stem]
            if (path:is-dir $dir) {
                builtin:put $dir
            }
        }
}
set edit:completion:arg-completer[{{cmd}}] = $__hview_h_complete~
