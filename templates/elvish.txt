# Jump to a directory using only keywords.
fn __peek_p {|@rest|
    if (builtin:eq [] $rest) {
        ls ~
    } elif (builtin:eq [-] $rest) {
        ls $oldpwd
    } elif (and ('builtin:==' (builtin:count $rest) 1) (path:is-dir &follow-symlink=$true $rest[0])) {
        ls $rest[0]
    } else {
        var path
        try {
            set path = (peek query --exclude $pwd -- $@rest)
        } catch {
        } else {
            ls $path
        }
    }
}
edit:add-var __peek_p~ $__peek_p~

# Jump to a directory using interactive search.
fn __peek_pi {|@rest|
    var path
    try {
        set path = (peek query --interactive -- $@rest)
    } catch {
    } else {
        ls $path
    }
}
edit:add-var __peek_pi~ $__peek_pi~

# Load completions.
{#-
  peek-based completions are currently not possible, because Elvish only prints
  a completion if the current token is a prefix of it.
#}
fn __peek_p_complete {|@rest|
    if (!= (builtin:count $rest) 2) {
        builtin:return
    }
    edit:complete-filename $rest[1] |
        builtin:each {|completion|
            var dir = $completion[stem]
            if (path:is-dir $dir) {
                builtin:put $dir
            }
        }
}
set edit:completion:arg-completer[{{cmd}}] = $__peek_p_complete~
