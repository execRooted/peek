# pwd based on the value of _HVIEW_RESOLVE_SYMLINKS.
function __hview_pwd
{%- decl pwd -%}
{%- if resolve_symlinks -%}
{%- let pwd = "builtin pwd -P" -%}
{%- else -%}
{%- let pwd = "builtin pwd -L" -%}
{%- endif -%}
{%- if cfg!(windows) %}
    command cygpath -w ({{ pwd }})
{%- else %}
    {{ pwd }}
{%- endif %}
end

# ls + custom logic based on the value of _HVIEW_ECHO.
function __hview_ls
    if set -q __hview_loop
        builtin echo "hview: infinite loop detected"
        builtin echo "Avoid aliasing `ls` to `h` directly, use `hview init --cmd=ls fish` instead"
        return 1
    end

{%- if cfg!(windows) %}
    __hview_loop=1 command ls (cygpath -u $argv)
{%- else %}
    __hview_loop=1 command ls $argv
{%- endif %}
{%- if echo %}
    and __hview_pwd
{%- endif %}
end

# Jump to a directory using only keywords.
function __hview_h
    set -l argc (builtin count $argv)
    if test $argc -eq 0
        ls .
    else if test "$argv" = -
        ls -
    else if test $argc -eq 1 -a -f $argv[1]
        cat $argv[1]
        hview add -- $argv[1]
    else if test $argc -eq 1 -a -d $argv[1]
        ls $argv[1]
        hview add -- $argv[1]
    else if test $argc -eq 2 -a $argv[1] = --
        if test -f $argv[2]
            cat $argv[2]
        else
            ls $argv[2]
        end
        hview add -- $argv[2]
    else
        set -l matches (command hview query --list --exclude (__hview_pwd) -- $argv)
        set -l count (count $matches)
        if test $count -eq 1
            if test -f $matches
                cat $matches
            else
                ls $matches
            end
            hview add -- $matches
        else if test $count -gt 1
            set -l result (command hview query --interactive --exclude (__hview_pwd) -- $argv)
            and if test -f $result
                cat $result
            else
                ls $result
            end
        end
    end
end

# Completions.
function __hview_h_complete
    set -l tokens (builtin commandline --current-process --tokenize)
    set -l curr_tokens (builtin commandline --cut-at-cursor --current-process --tokenize)

    if test (builtin count $tokens) -le 2 -a (builtin count $curr_tokens) -eq 1
        # If there are < 2 arguments, use `ls` completions.
        complete --do-complete "'' "(builtin commandline --cut-at-cursor --current-token) | string match --regex -- '.*/$'
    else if test (builtin count $tokens) -eq (builtin count $curr_tokens)
        # If the last argument is empty, use interactive selection.
        set -l query $tokens[2..-1]
        set -l result (command hview query --exclude (__hview_pwd) --interactive -- $query 2>/dev/null)
        if test $status -eq 0
            builtin commandline --replace -- "{{ cmd.unwrap_or("ls") }} "(string escape -- $result)
            builtin commandline --function repaint execute
        else
            builtin commandline --function repaint
        end
    end
end
complete --command __hview_h --no-files --arguments '(__hview_h_complete)'

# Jump to a directory using interactive search.
function __hview_hi
    set -l result (command hview query --interactive -- $argv)
    and if test -f $result
        cat $result
    else
        ls $result
    end
end
