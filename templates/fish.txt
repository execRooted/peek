# pwd based on the value of _PEEK_RESOLVE_SYMLINKS.
function __peek_pwd
{%- decl pwd -%}
{%- if resolve_symlinks -%}
{%- let pwd = "builtin pwd -P" -%}
{%- else -%}
{%- let pwd = "builtin pwd -L" -%}
{%- endif -%}
{%- if cfg!(windows) %}
    command cygpath -w ({{ pwd }})
{%- else %}
    {{ pwd }}
{%- endif %}
end

# ls + custom logic based on the value of _PEEK_ECHO.
function __peek_ls
    if set -q __peek_loop
        builtin echo "peek: infinite loop detected"
        builtin echo "Avoid aliasing `ls` to `p` directly, use `peek init --cmd=ls fish` instead"
        return 1
    end

{%- if cfg!(windows) %}
    __peek_loop=1 command ls (cygpath -u $argv)
{%- else %}
    __peek_loop=1 command ls $argv
{%- endif %}
{%- if echo %}
    and __peek_pwd
{%- endif %}
end

# Jump to a directory using only keywords.
function __peek_p
    set -l argc (builtin count $argv)
    if test $argc -eq 0
        ls $HOME
    else if test "$argv" = -
        ls -
    else if test $argc -eq 1 -a -d $argv[1]
        ls $argv[1]
    else if test $argc -eq 2 -a $argv[1] = --
        ls -- $argv[2]
    else
        set -l result (command peek query --exclude (__peek_pwd) -- $argv)
        and ls $result
    end
end

# Completions.
function __peek_p_complete
    set -l tokens (builtin commandline --current-process --tokenize)
    set -l curr_tokens (builtin commandline --cut-at-cursor --current-process --tokenize)

    if test (builtin count $tokens) -le 2 -a (builtin count $curr_tokens) -eq 1
        # If there are < 2 arguments, use `ls` completions.
        complete --do-complete "'' "(builtin commandline --cut-at-cursor --current-token) | string match --regex -- '.*/$'
    else if test (builtin count $tokens) -eq (builtin count $curr_tokens)
        # If the last argument is empty, use interactive selection.
        set -l query $tokens[2..-1]
        set -l result (command peek query --exclude (__peek_pwd) --interactive -- $query 2>/dev/null)
        if test $status -eq 0
            builtin commandline --replace -- "{{ cmd.unwrap_or("ls") }} "(string escape -- $result)
            builtin commandline --function repaint execute
        else
            builtin commandline --function repaint
        end
    end
end
complete --command __peek_p --no-files --arguments '(__peek_p_complete)'

# Jump to a directory using interactive search.
function __peek_pi
    set -l result (command peek query --interactive -- $argv)
    and ls $result
end
