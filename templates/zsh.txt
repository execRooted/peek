# pwd based on the value of _PEEK_RESOLVE_SYMLINKS.
function __peek_pwd() {
{%- decl pwd -%}
{%- if resolve_symlinks -%}
{%- let pwd = "\\builtin pwd -P" -%}
{%- else -%}
{%- let pwd = "\\builtin pwd -L" -%}
{%- endif -%}
{%- if cfg!(windows) %}
    \command cygpath -w "$({{ pwd }})"
{%- else %}
    {{ pwd }}
{%- endif %}
}

# ls + custom logic based on the value of _PEEK_ECHO.
function __peek_ls() {
    # shellcheck disable=SC2164
    \command ls -- "$@" {%- if echo %} && __peek_pwd {%- endif %}
}

# Hook configuration for peek.
#

# Hook to add new entries to the database.
function __peek_hook() {
    # shellcheck disable=SC2312
    \command peek add -- "$(__peek_pwd)"
}

# Initialize hook.
\builtin typeset -ga precmd_functions
\builtin typeset -ga chpwd_functions
# shellcheck disable=SC2034,SC2296
precmd_functions=("${(@)precmd_functions:#__peek_hook}")
# shellcheck disable=SC2034,SC2296
chpwd_functions=("${(@)chpwd_functions:#__peek_hook}")

{%- if hook == InitHook::Prompt %}
precmd_functions+=(__peek_hook)
{%- else if hook == InitHook::Pwd %}
chpwd_functions+=(__peek_hook)
{%- endif %}

# Report common issues.
function __peek_doctor() {
{%- if hook == InitHook::None %}
    return 0

{%- else %}
    [[ ${_PEEK_DOCTOR:-1} -ne 0 ]] || return 0
    [[ $- == *i* ]] || return 0

{%- if hook == InitHook::Prompt %}
    [[ ${precmd_functions[(Ie)__peek_hook]:-} -eq 0 ]] || return 0
{%- else if hook == InitHook::Pwd %}
    [[ ${chpwd_functions[(Ie)__peek_hook]:-} -eq 0 ]] || return 0
{%- endif %}

    _PEEK_DOCTOR=0
    \builtin printf '%s\n' \
        'peek: detected a possible configuration issue.' \
        'Please ensure that peek is initialized right at the end of your shell configuration file (usually ~/.zshrc).' \
        '' \
        'If the issue persists, consider filing an issue at:' \
        'https://github.com/execRooted/peek/issues' \
        '' \
        'Disable this message by setting _PEEK_DOCTOR=0.' \
        '' >&2
{%- endif %}
}

# Jump to a directory using only keywords.
function __peek_p() {
    __peek_doctor
    if [[ "$#" -eq 0 ]]; then
        ls ~
    elif [[ "$#" -eq 1 ]] && [[ "$1" = '-' ]]; then
        ls "${OLDPWD}"
    elif [[ "$#" -eq 1 ]] && { [[ "$1" =~ ^[-+][0-9]+$ ]] || (\builtin cd -q -- "$1") &>/dev/null; }; then
        ls "$1"
    elif [[ "$#" -eq 2 ]] && [[ "$1" = "--" ]]; then
        ls "$2"
    else
        \builtin local result
        # shellcheck disable=SC2312
        result="$(\command peek query --exclude "$(__peek_pwd)" -- "$@")" && ls "${result}"
    fi
}

# Jump to a directory using interactive search.
function __peek_pi() {
    __peek_doctor
    \builtin local result
    result="$(\command peek query --interactive -- "$@")" && ls "${result}"
}

# Completions.
if [[ -o zle ]]; then
    __peek_result=''

    function __peek_p_complete() {
        # Only show completions when the cursor is at the end of the line.
        # shellcheck disable=SC2154
        [[ "{{ "${#words[@]}" }}" -eq "${CURRENT}" ]] || return 0

        if [[ "{{ "${#words[@]}" }}" -eq 2 ]]; then
            # Show completions for local directories.
            _cd -/

        elif [[ "${words[-1]}" == '' ]]; then
            # Show completions for Space-Tab.
            # shellcheck disable=SC2086
            __peek_result="$(\command peek query --exclude "$(__peek_pwd || \builtin true)" --interactive -- ${words[2,-1]} 2>/dev/null)" || __peek_result=''

            # Set a result to ensure completion doesn't re-run
            compadd -Q -S "" -- ""

            # Bind '\e[0n' to helper function.
            \builtin bindkey '\e[0n' '__peek_p_complete_helper'
            # Sends query device status code, which results in a '\e[0n' being sent to console input.
            \builtin printf '\e[5n'

            # Report that the completion was successful, so that we don't fall back
            # to another completion function.
            return 0
        fi
    }

    function __peek_p_complete_helper() {
        if [[ -n "${__peek_result}" ]]; then
            # shellcheck disable=SC2034,SC2296
            BUFFER="{{ cmd.unwrap_or("ls") }} ${(q-)__peek_result}"
            __peek_result=''
            \builtin zle reset-prompt
            \builtin zle accept-line
        else
            \builtin zle reset-prompt
        fi
    }
    \builtin zle -N __peek_p_complete_helper
{%- if let Some(cmd) = cmd %}

    [[ "${+functions[compdef]}" -ne 0 ]] && \compdef __peek_p_complete {{ cmd }}
{%- endif %}
fi
