# pwd based on the value of _HVIEW_RESOLVE_SYMLINKS.
function __hview_pwd() {
{%- decl pwd -%}
{%- if resolve_symlinks -%}
{%- let pwd = "\\builtin pwd -P" -%}
{%- else -%}
{%- let pwd = "\\builtin pwd -L" -%}
{%- endif -%}
{%- if cfg!(windows) %}
    \command cygpath -w "$({{ pwd }})"
{%- else %}
    {{ pwd }}
{%- endif %}
}

# ls + custom logic based on the value of _HVIEW_ECHO.
function __hview_ls() {
    # shellcheck disable=SC2164
    \command ls -- "$@" {%- if echo %} && __hview_pwd {%- endif %}
}

# Wrapper for ls to automatically add listed directories to hview.
__hview_real_ls="ls"
if [[ $(type -t ls 2>/dev/null) == "alias" ]]; then
    __hview_real_ls="$(alias ls | sed "s/alias ls='\(.*\)'/\1/")"
fi

__hview_ls_wrapper() {
    eval "\command $__hview_real_ls \"\$@\"" || return $?
    if [[ $# -ge 1 ]] && [[ -d "${@: -1}" ]]; then
        \command hview add -- "${@: -1}"
    fi
}
alias ls=__hview_ls_wrapper

# Hook configuration for hview.
#

# Hook to add new entries to the database.
function __hview_hook() {
    # shellcheck disable=SC2312
    \command hview add -- "$(__hview_pwd)"
}

# Initialize hook.
\builtin typeset -ga precmd_functions
\builtin typeset -ga chpwd_functions
# shellcheck disable=SC2034,SC2296
precmd_functions=("${(@)precmd_functions:#__hview_hook}")
# shellcheck disable=SC2034,SC2296
chpwd_functions=("${(@)chpwd_functions:#__hview_hook}")

{%- if hook == InitHook::Prompt %}
precmd_functions+=(__hview_hook)
{%- else if hook == InitHook::Pwd %}
chpwd_functions+=(__hview_hook)
{%- endif %}

# Report common issues.
function __hview_doctor() {
{%- if hook == InitHook::None %}
    return 0

{%- else %}
    [[ ${_HVIEW_DOCTOR:-1} -ne 0 ]] || return 0
    [[ $- == *i* ]] || return 0

{%- if hook == InitHook::Prompt %}
    [[ ${precmd_functions[(Ie)__hview_hook]:-} -eq 0 ]] || return 0
{%- else if hook == InitHook::Pwd %}
    [[ ${chpwd_functions[(Ie)__hview_hook]:-} -eq 0 ]] || return 0
{%- endif %}

    _HVIEW_DOCTOR=0
    \builtin printf '%s\n' \
        'hview: detected a possible configuration issue.' \
        'Please ensure that hview is initialized right at the end of your shell configuration file (usually ~/.zshrc).' \
        '' \
        'If the issue persists, consider filing an issue at:' \
        'https://github.com/execRooted/hview/issues' \
        '' \
        'Disable this message by setting _HVIEW_DOCTOR=0.' \
        '' >&2
{%- endif %}
}

# Jump to a directory using only keywords.
function __hview_h() {
    __hview_doctor
    if [[ "$#" -eq 0 ]]; then
        \command ls ~
    elif [[ "$#" -eq 1 ]] && [[ "$1" = '-' ]]; then
        \command ls "${OLDPWD}"
    elif [[ "$#" -eq 1 ]] && { [[ "$1" =~ ^[-+][0-9]+$ ]] || (\builtin cd -q -- "$1") &>/dev/null; }; then
        \command ls "$1"
    elif [[ "$#" -eq 2 ]] && [[ "$1" = "--" ]]; then
        \command ls "$2"
    else
        \builtin local result
        # shellcheck disable=SC2312
        result="$(\command hview query --exclude "$(__hview_pwd)" -- "$@")" &&
            \command ls "${result}"
    fi
}

# Jump to a directory using interactive search.
function __hview_hi() {
    __hview_doctor
    \builtin local result
    result="$(\command hview query --interactive -- "$@")" && \command ls "${result}"
}

# Completions.
if [[ -o zle ]]; then
    __hview_result=''

    function __hview_h_complete() {
        # Only show completions when the cursor is at the end of the line.
        # shellcheck disable=SC2154
        [[ "{{ "${#words[@]}" }}" -eq "${CURRENT}" ]] || return 0

        if [[ "{{ "${#words[@]}" }}" -eq 2 ]]; then
            # Show completions for local directories.
            _cd -/

        elif [[ "${words[-1]}" == '' ]]; then
            # Show completions for Space-Tab.
            # shellcheck disable=SC2086
            __hview_result="$(\command hview query --exclude "$(__hview_pwd || \builtin true)" --interactive -- ${words[2,-1]} 2>/dev/null)" || __hview_result=''

            # Set a result to ensure completion doesn't re-run
            compadd -Q -S "" -- ""

            # Bind '\e[0n' to helper function.
            \builtin bindkey '\e[0n' '__hview_h_complete_helper'
            # Sends query device status code, which results in a '\e[0n' being sent to console input.
            \builtin printf '\e[5n'

            # Report that the completion was successful, so that we don't fall back
            # to another completion function.
            return 0
        fi
    }

    function __hview_h_complete_helper() {
        if [[ -n "${__hview_result}" ]]; then
            # shellcheck disable=SC2034,SC2296
            BUFFER="{{ cmd.unwrap_or("ls") }} ${(q-)__hview_result}"
            __hview_result=''
            \builtin zle reset-prompt
            \builtin zle accept-line
        else
            \builtin zle reset-prompt
        fi
    }
    \builtin zle -N __hview_h_complete_helper
{%- if let Some(cmd) = cmd %}

    [[ "${+functions[compdef]}" -ne 0 ]] && \compdef __hview_h_complete {{ cmd }}
{%- endif %}
fi
