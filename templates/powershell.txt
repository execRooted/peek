# Utility functions for peek.
#

# Call peek binary, returning the output as UTF-8.
function global:__peek_bin {
    $encoding = [Console]::OutputEncoding
    try {
        [Console]::OutputEncoding = [System.Text.Utf8Encoding]::new()
        $result = peek @args
        return $result
    } finally {
        [Console]::OutputEncoding = $encoding
    }
}

# pwd based on peek's format.
function global:__peek_pwd {
    $cwd = Microsoft.PowerShell.Management\Get-Location
    if ($cwd.Provider.Name -eq "FileSystem") {
        $cwd.ProviderPath
    }
}

# ls + custom logic based on the value of _PEEK_ECHO.
function global:__peek_ls($dir) {
    Microsoft.PowerShell.Management\Get-ChildItem -LiteralPath $dir
{%- if echo %}
    Microsoft.PowerShell.Utility\Write-Output $dir
{%- endif %}
}

# Hook to add new entries to the database.

function global:__peek_hook {
    $result = __peek_pwd
    if ($result -ne $global:__peek_oldpwd) {
        if ($null -ne $result) {
            peek add "--" $result

    }

# Jump to a directory using only keywords.
function global:__peek_p {
    if ($args.Length -eq 0) {
        __peek_ls $pwd
    }
    elseif ($args.Length -eq 1 -and ($args[0] -eq '-' -or $args[0] -eq '+')) {
        __peek_ls (Get-Location -Path $args[0])
    }
    elseif ($args.Length -eq 1 -and (Microsoft.PowerShell.Management\Test-Path -PathType Container -LiteralPath $args[0])) {
        __peek_ls $args[0]
    }
    elseif ($args.Length -eq 1 -and (Microsoft.PowerShell.Management\Test-Path -PathType Container -Path $args[0] )) {
        __peek_ls $args[0]
    }
    else {
        $result = __peek_pwd
        if ($null -ne $result) {
            $result = __peek_bin query --exclude $result "--" @args
        }
        else {
            $result = __peek_bin query "--" @args
        }
        if ($LASTEXITCODE -eq 0) {
            __peek_ls $result
        }
    }
}

# Jump to a directory using interactive search.
function global:__peek_pi {
    $result = __peek_bin query -i "--" @args
    if ($LASTEXITCODE -eq 0) {
        __peek_ls $result
    }
}
