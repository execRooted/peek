# Call hview binary, returning the output as UTF-8.
function global:__hview_bin {
    $encoding = [Console]::OutputEncoding
    try {
        [Console]::OutputEncoding = [System.Text.Utf8Encoding]::new()
        $result = hview @args
        return $result
    } finally {
        [Console]::OutputEncoding = $encoding
    }
}

# pwd based on hview's format.
function global:__hview_pwd {
    $cwd = Microsoft.PowerShell.Management\Get-Location
    if ($cwd.Provider.Name -eq "FileSystem") {
        $cwd.ProviderPath
    }
}

# ls + custom logic based on the value of _HVIEW_ECHO.
function global:__hview_ls($dir) {
    Microsoft.PowerShell.Management\Get-ChildItem -LiteralPath $dir
{%- if echo %}
    Microsoft.PowerShell.Utility\Write-Output $dir
{%- endif %}
}

function global:__hview_hook {
    $result = __hview_pwd
    if ($null -ne $result) {
        hview add "--" $result
    }
}
{%- else if hook == InitHook::Pwd -%}
# Hook to add new entries to the database.
$global:__hview_oldpwd = __hview_pwd
function global:__hview_hook {
    $result = __hview_pwd
    if ($result -ne $global:__hview_oldpwd) {
        if ($null -ne $result) {
            hview add "--" $result
        }
        $global:__hview_oldpwd = $result
    }
}

    $global:__hview_prompt_old = $function:prompt

    function global:prompt {
        if ($null -ne $__hview_prompt_old) {
            & $__hview_prompt_old
        }
        $null = __hview_hook
    }

# Jump to a directory using only keywords.
function global:__hview_h {
    if ($args.Length -eq 0) {
        __hview_ls $pwd
    }
    elseif ($args.Length -eq 1 -and ($args[0] -eq '-' -or $args[0] -eq '+')) {
        __hview_ls (Get-Location -Path $args[0])
    }
    elseif ($args.Length -eq 1 -and (Microsoft.PowerShell.Management\Test-Path -PathType Container -LiteralPath $args[0])) {
        __hview_ls $args[0]
        hview add "--" $args[0]
    }
    elseif ($args.Length -eq 1 -and (Microsoft.PowerShell.Management\Test-Path -PathType Container -Path $args[0] )) {
        __hview_ls $args[0]
        hview add "--" $args[0]
    }
    elseif ($args.Length -eq 1 -and (Microsoft.PowerShell.Management\Test-Path -PathType Leaf -LiteralPath $args[0])) {
        Microsoft.PowerShell.Utility\Get-Content -LiteralPath $args[0]
        hview add "--" $args[0]
    }
    elseif ($args.Length -eq 1 -and (Microsoft.PowerShell.Management\Test-Path -PathType Leaf -Path $args[0] )) {
        Microsoft.PowerShell.Utility\Get-Content -LiteralPath $args[0]
        hview add "--" $args[0]
    }
    else {
        $result = __hview_pwd
        if ($null -ne $result) {
            $list = __hview_bin query --list --exclude $result "--" @args
        }
        else {
            $list = __hview_bin query --list "--" @args
        }
        if ($LASTEXITCODE -eq 0 -and $list) {
            $count = ($list -split "`n").Count
            if ($count -eq 1) {
                if ((Microsoft.PowerShell.Management\Get-Item -LiteralPath $list).PSIsContainer) {
                    __hview_ls $list
                } else {
                    Microsoft.PowerShell.Utility\Get-Content -LiteralPath $list
                }
                hview add "--" $list
            }
            else {
                if ($null -ne $result) {
                    $result = __hview_bin query --interactive --exclude $result "--" @args
                }
                else {
                    $result = __hview_bin query --interactive "--" @args
                }
                if ($LASTEXITCODE -eq 0) {
                    if ((Microsoft.PowerShell.Management\Get-Item -LiteralPath $result).PSIsContainer) {
                        __hview_ls $result
                    } else {
                        Microsoft.PowerShell.Utility\Get-Content -LiteralPath $result
                    }
                }
            }
        }
    }
}

# Jump to a directory using interactive search.
function global:__hview_hi {
    $result = __hview_bin query -i "--" @args
    if ($LASTEXITCODE -eq 0) {
        if ((Microsoft.PowerShell.Management\Get-Item -LiteralPath $result).PSIsContainer) {
            __hview_ls $result
        } else {
            Microsoft.PowerShell.Utility\Get-Content -LiteralPath $result
        }
    }
}
