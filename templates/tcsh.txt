# Hook configuration for hview.
#
{%- if hook != InitHook::None %}

# Hook to add new entries to the database.
{%- if hook == InitHook::Prompt %}
alias __hview_hook 'hview add -- "`{{ pwd_cmd }}`"'

{%- else if hook == InitHook::Pwd %}
set __hview_pwd_old = `{{ pwd_cmd }}`
alias __hview_hook 'set __hview_pwd_tmp = "`{{ pwd_cmd }}`"; test "$__hview_pwd_tmp" != "$__hview_pwd_old" && hview add -- "$__hview_pwd_tmp"; set __hview_pwd_old = "$__hview_pwd_tmp"'
{%- endif %}

# Initialize hook.
set __hview_precmd = "`alias precmd`"
if ( "$__hview_precmd" !~ *__hview_hook* ) then
    if ( "$__hview_precmd" == "" ) then
        alias precmd '__hview_hook'
    else
        alias precmd "$__hview_precmd"';__hview_hook'
    endif
endif
unset __hview_precmd

{%- endif %}

# When using hview with --no-cmd, alias these internal functions as desired.
#

# Jump to a directory using only keywords.
alias __hview_h 'set __hview_args = (\!*)\
if ("$#__hview_args" == 0) then\
    ls .\
else\
    if ("$#__hview_args" == 1 && "$__hview_args[1]" == "-") then\
        ls -\
    else if ("$#__hview_args" == 1 && -d "$__hview_args[1]") then\
        ls "$__hview_args[1]"\
    else\
        set __hview_pwd = `{{ pwd_cmd }}`\
        set __hview_result = "`hview query --exclude '"'"'$__hview_pwd'"'"' -- $__hview_args`" && ls "$__hview_result"\
    endif\
endif'

# Jump to a directory using interactive search.
alias __hview_hi 'set __hview_args = (\!*)\
set __hview_pwd = `{{ pwd_cmd }}`\
set __hview_result = "`hview query --exclude '"'"'$__hview_pwd'"'"' --interactive -- $__hview_args`" && ls "$__hview_result"'
#

{%- match cmd %}
{%- when Some with (cmd) %}

alias {{cmd}} __hview_h
alias {{cmd}}i __hview_hi
