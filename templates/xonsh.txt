def __hview_bin() -> str:
    """Finds and returns the location of the hview binary."""
    hview = typing.cast(str, xonsh.environ.locate_binary("hview"))
    if hview is None:
        hview = "hview"
    return hview


def __hview_env() -> dict[str, str]:
    """Returns the current environment."""
    return builtins.__xonsh__.env.detype()  # type: ignore  # pylint:disable=no-member


def __hview_pwd() -> str:
    """pwd based on the value of _HVIEW_RESOLVE_SYMLINKS."""

{%- else %}
    pwd = __hview_env().get("PWD")
    if pwd is None:
        raise RuntimeError("$PWD not found")
{%- endif %}
    return pwd


def __hview_ls(path: str | bytes | None = None) -> None:
    """ls + custom logic based on the value of _HVIEW_ECHO."""
    if path is None:
        args = []
    elif isinstance(path, bytes):
        args = [path.decode("utf-8")]
    else:
        args = [path]
    _, exc, _ = xonsh.dirstack.ls(args)
    if exc is not None:
        raise RuntimeError(exc)

class HviewSilentException(Exception):
    """Exit without complaining."""


def __hview_errhandler(
    func: typing.Callable[[list[str]], None],
) -> typing.Callable[[list[str]], int]:
    """Print exception and exit with error code 1."""

    def wrapper(args: list[str]) -> int:
        try:
            func(args)
            return 0
        except HviewSilentException:
            return 1
        except Exception as exc:  # pylint: disable=broad-except
            print(f"hview: {exc}", file=sys.stderr)
            return 1

    return wrapper

@__hview_errhandler
def __hview_h(args: list[str]) -> None:
    """Jump to a directory using only keywords."""
    if args == []:
        __hview_ls()
    elif args == ["-"]:
        __hview_ls("-")
    elif len(args) == 1 and os.path.isdir(args[0]):
        __hview_ls(args[0])
        hview = __hview_bin()
        subprocess.run([hview, "add", "--", args[0]], check=False)
    elif len(args) == 1 and os.path.isfile(args[0]):
        with open(args[0]) as f:
            sys.stdout.write(f.read())
        hview = __hview_bin()
        subprocess.run([hview, "add", "--", args[0]], check=False)
    else:
        try:
            hview = __hview_bin()
            cmd = subprocess.run(
                [hview, "query", "--list", "--exclude", __hview_pwd(), "--"] + args,
                check=True,
                env=__hview_env(),
                stdout=subprocess.PIPE,
            )
            matches = cmd.stdout.decode("utf-8").strip()
            if matches:
                count = len(matches.splitlines())
                if count == 1:
                    if os.path.isfile(matches):
                        with open(matches) as f:
                            sys.stdout.write(f.read())
                    else:
                        __hview_ls(matches)
                    subprocess.run([hview, "add", "--", matches], check=False)
                else:
                    cmd = subprocess.run(
                        [hview, "query", "--interactive", "--exclude", __hview_pwd(), "--"] + args,
                        check=True,
                        env=__hview_env(),
                        stdout=subprocess.PIPE,
                    )
                    result = cmd.stdout.decode("utf-8")[:-1]
                    if os.path.isfile(result):
                        with open(result) as f:
                            sys.stdout.write(f.read())
                    else:
                        __hview_ls(result)
        except subprocess.CalledProcessError as exc:
            raise HviewSilentException() from exc


@__hview_errhandler
def __hview_hi(args: list[str]) -> None:
    """Jump to a directory using interactive search."""
    try:
        hview = __hview_bin()
        cmd = subprocess.run(
            [hview, "query", "-i", "--"] + args,
            check=True,
            env=__hview_env(),
            stdout=subprocess.PIPE,
        )
    except subprocess.CalledProcessError as exc:
        raise HviewSilentException() from exc

    result = cmd.stdout.decode("utf-8")[:-1]
    if os.path.isfile(result):
        with open(result) as f:
            sys.stdout.write(f.read())
    else:
        __hview_ls(result)
