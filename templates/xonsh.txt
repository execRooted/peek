# Utility functions for peek.
#


def __peek_bin() -> str:
    """Finds and returns the location of the peek binary."""
    peek = typing.cast(str, xonsh.environ.locate_binary("peek"))
    if peek is None:
        peek = "peek"
    return peek


def __peek_env() -> dict[str, str]:
    """Returns the current environment."""
    return builtins.__xonsh__.env.detype()  # type: ignore  # pylint:disable=no-member


def __peek_pwd() -> str:
    """pwd based on the value of _PEEK_RESOLVE_SYMLINKS."""

{%- else %}
    pwd = __peek_env().get("PWD")
    if pwd is None:
        raise RuntimeError("$PWD not found")
{%- endif %}
    return pwd


def __peek_ls(path: str | bytes | None = None) -> None:
    """ls + custom logic based on the value of _PEEK_ECHO."""
    if path is None:
        args = []
    elif isinstance(path, bytes):
        args = [path.decode("utf-8")]
    else:
        args = [path]
    _, exc, _ = xonsh.dirstack.ls(args)
    if exc is not None:
        raise RuntimeError(exc)

class PeekSilentException(Exception):
    """Exit without complaining."""


def __peek_errhandler(
    func: typing.Callable[[list[str]], None],
) -> typing.Callable[[list[str]], int]:
    """Print exception and exit with error code 1."""

    def wrapper(args: list[str]) -> int:
        try:
            func(args)
            return 0
        except PeekSilentException:
            return 1
        except Exception as exc:  # pylint: disable=broad-except
            print(f"peek: {exc}", file=sys.stderr)
            return 1

    return wrapper

@__peek_errhandler
def __peek_p(args: list[str]) -> None:
    """Jump to a directory using only keywords."""
    if args == []:
        __peek_ls()
    elif args == ["-"]:
        __peek_ls("-")
    elif len(args) == 1 and os.path.isdir(args[0]):
        __peek_ls(args[0])
    else:
        try:
            peek = __peek_bin()
            cmd = subprocess.run(
                [peek, "query", "--exclude", __peek_pwd(), "--"] + args,
                check=True,
                env=__peek_env(),
                stdout=subprocess.PIPE,
            )
        except subprocess.CalledProcessError as exc:
            raise PeekSilentException() from exc

        result = cmd.stdout[:-1]
        __peek_ls(result)


@__peek_errhandler
def __peek_pi(args: list[str]) -> None:
    """Jump to a directory using interactive search."""
    try:
        peek = __peek_bin()
        cmd = subprocess.run(
            [peek, "query", "-i", "--"] + args,
            check=True,
            env=__peek_env(),
            stdout=subprocess.PIPE,
        )
    except subprocess.CalledProcessError as exc:
        raise PeekSilentException() from exc

    result = cmd.stdout[:-1]
    __peek_ls(result)
